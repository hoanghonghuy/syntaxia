  - Use 我 / 你 / 他 / 她
  - Form plurals with 们 and ask for a 名字
vocab:
  - { hanzi: "我", pinyin: "wǒ", gloss: "I / me" }
  - { hanzi: "你", pinyin: "nǐ", gloss: "you" }
  - { hanzi: "他", pinyin: "tā", gloss: "he / him" }
  - { hanzi: "她", pinyin: "tā", gloss: "she / her" }
  - { hanzi: "我们", pinyin: "wǒ men", gloss: "we / us" }
  - { hanzi: "你们", pinyin: "nǐ men", gloss: "you (plural)" }
  - { hanzi: "人", pinyin: "rén", gloss: "person / people" }
  - { hanzi: "名字", pinyin: "míng zi", gloss: "name" }
steps:
  - type: dialogue
    lines:
      - { speaker: "A", text: "你好！你叫什么名字？", reading: "nǐ hǎo! nǐ jiào shén me míng zi?" }
      - { speaker: "B", text: "我叫小明。你呢？", reading: "wǒ jiào xiǎo míng. nǐ ne?" }
      - { speaker: "A", text: "我叫小红。我们是同学。", reading: "wǒ jiào xiǎo hóng. wǒ men shì tóng xué." }
  - type: tip
    title: "我 / 你 + 名字"
    body: "我叫… means “I am called…”. 们 after a pronoun marks plural: 我 → 我们, 你 → 你们. 他 and 她 share pinyin tā — the characters differ."
  - type: teach
    items:
      - { form: "我", reading: "wǒ", gloss: "I / me", example: "我是学生。" }
      - { form: "你", reading: "nǐ", gloss: "you", example: "你好吗？" }
      - { form: "名字", reading: "míng zi", gloss: "name", example: "你叫什么名字？" }
      - { form: "我们", reading: "wǒ men", gloss: "we / us", example: "我们是同学。" }
      - { form: "他", reading: "tā", gloss: "he / him", example: "他是老师。" }
  - type: practice
    kind: mcq
    prompt: "Which means “we / us”?"
    choices: ["我们", "你们", "名字"]
    answer: "我们"
    hints:
      - "们 after a pronoun often marks plural."
  - type: practice
    kind: mcq
    prompt: "In the dialogue, how does B ask A's name back?"
    choices: ["你呢？", "谢谢！", "再见！"]
    answer: "你呢？"
  - type: checkpoint
    items:
      - kind: mcq
        prompt: "Which means “hello”?"
        choices: ["你好", "名字", "我们"]
        answer: "你好"
      - kind: mcq
        prompt: "Which means “name”?"
        choices: ["名字", "人", "她"]
        answer: "名字"
      - kind: fill_blank
        prompt: "Type the character for “I / me”."
        answer: "我"
exercise:
  type: mcq
  prompt: "Which means “we / us”?"
  choices: ["我们", "你们", "名字"]
  answer: "我们"
  hints:
    - "们 after a pronoun often marks plural."
    - "我 + 们 → 我们."
